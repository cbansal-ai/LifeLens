Email (message)
│
├── id
├── threadId
├── snippet
└── payload
      │
      ├── headers

      │     ├── From

      │     ├── To

      │     ├── Subject

      │     ├── Date

      │     └── ...

      │

      ├── body (if single-part email)

      │

      └── parts (if multipart email)

            │

            ├── Part 1 (text/plain)

            │      └── body

            │

            ├── Part 2 (text/html)

            │      └── body

            │

            └── Part 3 (attachment)

                   └── body


create table events (

    id bigint generated always as identity primary key,

    account_email text,

    company text,

    event_type text,

    title text,

    event_date timestamp,

    summary text,

    raw_json jsonb,

    created_at timestamp default now()

);
